// Requires PowerSync Swift SDK 1.17.0 or later.
import PowerSync

/// Connects `db` for downloads only. To upload local writes as well, add the mutation uploader and use its
/// connect() instead.
func connectDownloadOnly(db: PowerSyncDatabaseProtocol) async throws {
    try await db.connect(
        endpoint: "[Your PowerSync instance URL or self-hosted endpoint]",
        authenticator: MyAuthenticator(),
        options: ConnectOptions()
    )
}

/// Provides the token that the SDK sends to your PowerSync instance. The token is cached until it is about to
/// expire or is rejected.
/// An actor, because the SDK requires a Sendable authenticator and this one caches a token.
actor MyAuthenticator: Authenticator {
    private var token: String?

    func resolveCredentials() async throws -> String {
        if let token {
            return token
        }
        let token = try await fetchAuthToken()
        self.token = token
        return token
    }

    /// The SDK calls this before the token expires and when the PowerSync Service rejects it. Keep the cache even if
    /// your auth provider's SDK refreshes sessions itself: the PowerSync SDK isn't guaranteed to cache tokens.
    func invalidateCredentials() async {
        token = nil
    }

    /// Replace this with your auth provider's token retrieval. See https://docs.powersync.com/configuration/auth/overview
    /// Supabase and Firebase tokens can be used as they are:
    /// - https://docs.powersync.com/configuration/auth/supabase-auth
    /// - https://docs.powersync.com/configuration/auth/firebase-auth
    /// To get started without an auth provider, use a development token:
    /// https://docs.powersync.com/configuration/auth/development-tokens
    private func fetchAuthToken() async throws -> String {
        return "An authentication token"
    }
}
