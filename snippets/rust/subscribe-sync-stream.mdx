use serde_json::json;

// Subscribe to a stream with parameters
let sub = db
    .sync_stream("list_todos", Some(&json!({ "list_id": "abc123" })))
    .subscribe()
    .await?;

// Wait for the initial data to sync
sub.wait_for_first_sync().await;
