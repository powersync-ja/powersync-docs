```kotlin PowerSync.kt
// Requires PowerSync Kotlin SDK 1.16.1 or later, with the Supabase integration:
//   implementation("com.powersync:connector-supabase:$powersyncVersion")
import com.powersync.PowerSyncDatabase
import com.powersync.connector.supabase.SupabaseAuthenticator
import com.powersync.connector.supabase.SupabaseUploader
import io.github.jan.supabase.SupabaseClient
import io.github.jan.supabase.auth.Auth
import io.github.jan.supabase.auth.auth
import io.github.jan.supabase.auth.status.SessionStatus
import io.github.jan.supabase.createSupabaseClient
import io.github.jan.supabase.postgrest.Postgrest
import kotlinx.coroutines.flow.distinctUntilChangedBy
import kotlinx.coroutines.flow.filter

/** Creates the Supabase client that your app signs in with. PowerSync needs its Auth and Postgrest plugins. */
fun createSupabase(supabaseUrl: String, supabaseAnonKey: String): SupabaseClient =
    createSupabaseClient(supabaseUrl, supabaseAnonKey) {
        install(Auth)
        install(Postgrest)
    }

/**
 * Connects `db` while a user is signed in to Supabase, and disconnects and clears local data when they sign out.
 * Call this from a coroutine that lives as long as the database, for example at app startup.
 *
 * SupabaseAuthenticator sends the access token of the current Supabase session, which the Supabase client
 * refreshes. SupabaseUploader uploads writes through PostgREST, so your RLS policies apply to them. It discards
 * a transaction that fails with a data exception (class 22), an integrity constraint violation (class 23), or
 * insufficient privilege (42501), and retries other errors. To change this, override handleError().
 */
suspend fun syncWithSupabase(db: PowerSyncDatabase, supabase: SupabaseClient, powerSyncUrl: String) {
    val authenticator = SupabaseAuthenticator(supabase)
    val uploader = SupabaseUploader(supabase)

    supabase.auth.sessionStatus
        .filter { it is SessionStatus.Authenticated || it is SessionStatus.NotAuthenticated }
        // The Supabase client reports Authenticated again after each token refresh. The SDK requests the new
        // token itself, so only react when the user signs in or out.
        .distinctUntilChangedBy { it is SessionStatus.Authenticated }
        .collect { status ->
            if (status is SessionStatus.Authenticated) {
                db.connect(endpoint = powerSyncUrl, authenticator = authenticator, uploader = uploader)
            } else {
                db.disconnectAndClear()
            }
        }
}
```
