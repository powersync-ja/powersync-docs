```rust
use std::pin::pin;
use futures_lite::StreamExt;
use powersync::{PowerSyncDatabase, error::PowerSyncError};
use rusqlite::params;

struct PendingList {
    id: String,
    name: String,
}

async fn watch_pending_lists(db: &PowerSyncDatabase) -> Result<(), PowerSyncError> {
    let stream = db.watch_statement(
        "SELECT id, name FROM lists WHERE state = ?".to_string(),
        params!["pending"],
        |stmt, params| {
            let mut rows = stmt.query(params)?;
            let mut lists = vec![];

            while let Some(row) = rows.next()? {
                lists.push(PendingList {
                    id: row.get("id")?,
                    name: row.get("name")?,
                });
            }

            Ok(lists)
        },
    );
    let mut stream = pin!(stream);

    // Run this function in a separate async task because the stream keeps listening.
    while let Some(lists) = stream.try_next().await? {
        for list in lists {
            println!("{}: {}", list.id, list.name);
        }
    }
    Ok(())
}
```
