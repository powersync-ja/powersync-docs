// Requires powersync 2.6.0 or later.
import 'package:flutter/foundation.dart';
import 'package:path/path.dart';
import 'package:path_provider/path_provider.dart';
import 'package:powersync/powersync.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import './app_config.dart';
import './models/schema.dart';
import './supabase.dart';

/// Use Supabase for authentication.
// Gives the SDK the signed-in user's Supabase access token. openDatabase() below connects for downloads only with:
//   db.connect(connector: ConnectRequest.downloadOnly(endpoint: AppConfig.powersyncUrl, authenticator: authenticator));
class SupabaseAuthenticator implements Authenticator {
  Future<void>? _refreshFuture;

  String? _cachedToken;

  /// Returns the Supabase access token for sync. Keep the cache: the SDK isn't
  /// guaranteed to cache tokens. The SDK calls [invalidateCredentials] only
  /// when the token has expired or the PowerSync Service rejects it, and this
  /// method then waits for the session refresh that it starts.
  /// `currentSession` doesn't refresh an expired session by itself, so an
  /// expired session is refreshed here before its token is used.
  @override
  Future<String> resolveCredentials(Future<void> abortSignal) async {
    if (_cachedToken case String token) {
      return token;
    }

    // Wait for pending session refresh if any
    await _refreshFuture;

    // Use Supabase token for PowerSync
    final auth = Supabase.instance.client.auth;
    var session = auth.currentSession;
    if (session == null) {
      // Not logged in
      throw StateError('Not logged in');
    }
    if (session.isExpired) {
      // Throws if the refresh fails. The SDK then retries after a delay.
      session = (await auth.refreshSession()).session ?? session;
    }

    // Use the access token to authenticate against PowerSync
    return _cachedToken = session.accessToken;
  }

  /// Clears the cached token after a session change.
  void clearCachedToken() {
    _cachedToken = null; // A new session has a new token.
  }

  @override
  void invalidateCredentials() {
    _cachedToken = null;

    // Trigger a session refresh if auth fails on PowerSync.
    // Generally, sessions should be refreshed automatically by Supabase.
    // However, in some cases it can be a while before the session refresh is
    // retried. We attempt to trigger the refresh as soon as we get an auth
    // failure on PowerSync.
    //
    // This could happen if the device was offline for a while and the session
    // expired, and nothing else attempt to use the session it in the meantime.
    //
    // Timeout the refresh call to avoid waiting for long retries,
    // and ignore any errors. resolveCredentials retries the refresh if the
    // session is still expired.
    _refreshFuture = Supabase.instance.client.auth
        .refreshSession()
        .timeout(const Duration(seconds: 5))
        .then((response) => null, onError: (error) => null);
  }
}

/// Global reference to the database
late final PowerSyncDatabase db;

bool isLoggedIn() {
  return Supabase.instance.client.auth.currentSession?.accessToken != null;
}

/// id of the user currently logged in
String? getUserId() {
  return Supabase.instance.client.auth.currentSession?.user.id;
}

Future<String> getDatabasePath() async {
  const dbFilename = 'powersync-demo.db';
  // getApplicationSupportDirectory is not supported on Web
  if (kIsWeb) {
    return dbFilename;
  }
  final dir = await getApplicationSupportDirectory();
  return join(dir.path, dbFilename);
}

Future<void> openDatabase() async {
  // Open the local database
  db = PowerSyncDatabase(
      schema: schema, path: await getDatabasePath(), logger: attachedLogger);
  await db.initialize();

  await loadSupabase();

  SupabaseAuthenticator? currentAuthenticator;

  if (isLoggedIn()) {
    // If the user is already logged in, connect immediately.
    // Otherwise, connect once logged in.
    currentAuthenticator = SupabaseAuthenticator();
    db.connect(
        connector: ConnectRequest.downloadOnly(
            endpoint: AppConfig.powersyncUrl,
            authenticator: currentAuthenticator));
  }

  Supabase.instance.client.auth.onAuthStateChange.listen((data) async {
    final AuthChangeEvent event = data.event;
    if (event == AuthChangeEvent.signedIn) {
      // Connect to PowerSync when the user is signed in
      currentAuthenticator = SupabaseAuthenticator();
      db.connect(
          connector: ConnectRequest.downloadOnly(
              endpoint: AppConfig.powersyncUrl,
              authenticator: currentAuthenticator!));
    } else if (event == AuthChangeEvent.signedOut) {
      // Implicit sign out - disconnect, but don't delete data
      currentAuthenticator = null;
      await db.disconnect();
    } else if (event == AuthChangeEvent.tokenRefreshed) {
      // Supabase token refreshed - PowerSync uses the new token on its next request.
      currentAuthenticator?.clearCachedToken();
    }
  });
}

/// Explicit sign out - clear database and log out.
Future<void> logout() async {
  await Supabase.instance.client.auth.signOut();
  await db.disconnectAndClear();
}
