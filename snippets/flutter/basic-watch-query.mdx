```dart
import 'package:flutter/material.dart';
import 'package:powersync/powersync.dart';

class PendingLists extends StatelessWidget {
  final PowerSyncDatabase db;

  const PendingLists({super.key, required this.db});

  @override
  Widget build(BuildContext context) {
    return StreamBuilder(
      stream: db.watch('SELECT id, name FROM lists WHERE state = ?', ['pending']),
      builder: (context, snapshot) {
        if (snapshot.hasError) {
          return const Text('Could not load lists.');
        }
        if (!snapshot.hasData) {
          return const Center(child: CircularProgressIndicator());
        }

        final lists = snapshot.requireData;
        return ListView(
          children: [
            for (final list in lists)
              ListTile(
                key: ValueKey(list['id']),
                title: Text(list['name'] as String),
              ),
          ],
        );
      },
    );
  }
}
```
