// Requires PowerSync.Common 1.0.1 or later.
using System.Threading.Tasks;
using PowerSync.Common.Client;
using PowerSync.Common.Client.Connection;

/// <summary>
/// Provides the token that the SDK sends to your PowerSync instance. The token is cached until it is about to
/// expire or is rejected.
/// </summary>
public class MyAuthenticator : IAuthenticator
{
    /// <summary>
    /// Connects <paramref name="db"/> for downloads only. To upload local writes as well, add the mutation uploader
    /// and use its Connect() instead.
    /// </summary>
    public static async Task ConnectDownloadOnly(IPowerSyncDatabase db)
    {
        await db.Connect(
            "[Your PowerSync instance URL or self-hosted endpoint]",
            new MyAuthenticator(),
            null);
    }

    private string? _token;

    public async Task<string> ResolveCredentials()
    {
        return _token ??= await FetchAuthToken();
    }

    /// <summary>
    /// The SDK calls this before the token expires and when the PowerSync Service rejects it. Keep the cache even if
    /// your auth provider's SDK refreshes sessions itself: the PowerSync SDK isn't guaranteed to cache tokens.
    /// </summary>
    public void InvalidateCredentials()
    {
        _token = null;
    }

    /// <summary>
    /// Replace this with your auth provider's token retrieval. See https://docs.powersync.com/configuration/auth/overview
    /// Supabase and Firebase tokens can be used as they are:
    /// - https://docs.powersync.com/configuration/auth/supabase-auth
    /// - https://docs.powersync.com/configuration/auth/firebase-auth
    /// To get started without an auth provider, use a development token:
    /// https://docs.powersync.com/configuration/auth/development-tokens
    /// </summary>
    private Task<string> FetchAuthToken()
    {
        return Task.FromResult("An authentication token");
    }
}
