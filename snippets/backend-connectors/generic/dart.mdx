```dart lib/powersync.dart
// Uploads local writes to the PowerSync reference write API, one transaction per request.
// Set up the write API: https://docs.powersync.com/handling-writes/write-path/reference-write-implementation
// If your backend has a different API, change the request and response handling in _postTransaction() to match
// it, or have a coding agent write uploadData() for your API:
// https://docs.powersync.com/handling-writes/write-path/existing-api#prompt-for-your-coding-agent
// For several transactions per request and client-side backoff, see the full reference connector:
// https://github.com/powersync-ja/powersync-reference-write-implementation/blob/main/example-client/src/PowersyncConnector.singlefile.ts
import 'dart:convert';

import 'package:http/http.dart' as http; // Add the http package: https://pub.dev/packages/http
import 'package:powersync/powersync.dart';

/// Base URL of the write API. On a phone or emulator, `localhost` refers to the device itself.
const backendUrl = 'http://localhost:6060';

/// Abort a write API request after this long.
const requestTimeout = Duration(seconds: 30);

/// Consecutive retryable errors for one transaction before it is passed to onFatalTransaction().
const retryMaxAttempts = 10;

class MyBackendConnector extends PowerSyncBackendConnector {
  String? _authToken;
  Object? _retryKey;
  int _retryCount = 0;

  @override
  Future<PowerSyncCredentials?> fetchCredentials() async {
    return PowerSyncCredentials(
      endpoint: '[Your PowerSync instance URL or self-hosted endpoint]',
      token: await _getAuthToken(),
    );
  }

  /// Uploads the oldest queued transaction. The SDK calls this after local writes, on (re)connect and on
  /// keepalives, and keeps calling it until the upload queue is empty. Throwing keeps the transaction queued,
  /// and the SDK retries after a delay.
  /// See https://docs.powersync.com/configuration/app-backend/client-side-integration#when-uploaddata-is-called
  @override
  Future<void> uploadData(PowerSyncDatabase database) async {
    final transaction = await database.getNextCrudTransaction();
    if (transaction == null) {
      return;
    }

    Map<String, dynamic> result;
    try {
      result = await _postTransaction(transaction);
    } on AuthenticationException {
      // Fetch a new token before the next attempt. Like other transport errors, this doesn't count toward
      // retryMaxAttempts.
      _authToken = null;
      rethrow;
    }

    result = _applyRetryLimit(transaction, result);

    switch (result['status']) {
      case 'success':
        break;
      case 'retryable_error':
        throw Exception(result['message'] ?? 'Retryable error');
      case 'fatal_error':
        if (result['requires_client_handling'] == true &&
            await onFatalTransaction(transaction, result) == FatalDecision.retain) {
          throw Exception('Transaction retained after ${result['failed_operation']['error_code']}');
        }
        // A fatal error that the write API already handled (for example, with a dead-letter hook), or one that
        // onFatalTransaction() chose to discard.
        break;
      default:
        throw Exception('Unexpected transaction result: ${result['status']}');
    }
    await transaction.complete();
  }

  /// Called for fatal errors that the write API leaves to the client, and when a transaction reaches
  /// retryMaxAttempts. [FatalDecision.retain] keeps the transaction queued, which blocks later uploads, and the
  /// SDK calls this again on its next attempt. Return [FatalDecision.complete] to discard the transaction, for
  /// example after telling the user.
  /// See https://github.com/powersync-ja/powersync-reference-write-implementation/blob/main/docs/error-handling.md
  Future<FatalDecision> onFatalTransaction(CrudTransaction transaction, Map<String, dynamic> result) async {
    print('Transaction needs client handling: ${result['failed_operation']['error_code']} ${result['message']}');
    return FatalDecision.retain;
  }

  /// Returns the token for sync and for the write API, which accepts the same JWTs as your PowerSync instance.
  /// The token is cached until the write API rejects it. If your auth provider's SDK refreshes sessions itself,
  /// return `_fetchAuthToken()` directly instead.
  Future<String> _getAuthToken() async {
    return _authToken ??= await _fetchAuthToken();
  }

  /// Replace this with your auth provider's token retrieval. See https://docs.powersync.com/configuration/auth/overview
  /// Supabase and Firebase tokens can be used as they are:
  /// - https://docs.powersync.com/configuration/auth/supabase-auth
  /// - https://docs.powersync.com/configuration/auth/firebase-auth
  /// To use a development token with the write API, see
  /// https://github.com/powersync-ja/powersync-reference-write-implementation/blob/main/docs/auth/development-tokens.md
  Future<String> _fetchAuthToken() async {
    return 'An authentication token';
  }

  /// Sends one transaction to POST /api/data and returns its result. The request and response shapes are defined
  /// by the write API's OpenAPI spec:
  /// https://github.com/powersync-ja/powersync-reference-write-implementation/blob/main/backend/shared/powersync-reference-write-api.openapi.yaml
  Future<Map<String, dynamic>> _postTransaction(CrudTransaction transaction) async {
    final body = {
      'transactions': [
        {
          if (transaction.transactionId != null) 'transaction_id': transaction.transactionId,
          'crud': [
            for (final op in transaction.crud)
              {
                'id': op.id,
                'op': op.op.json, // PUT, PATCH or DELETE
                'table': op.table,
                if (op.transactionId != null) 'transaction_id': op.transactionId,
                if (op.opData != null) 'op_data': op.opData, // Changed columns for PUT and PATCH
              },
          ],
        },
      ],
    };

    final response = await http
        .post(
          Uri.parse('$backendUrl/api/data'),
          headers: {
            'Content-Type': 'application/json',
            'Authorization': 'Bearer ${await _getAuthToken()}',
          },
          body: jsonEncode(body),
        )
        .timeout(requestTimeout);

    if (response.statusCode < 200 || response.statusCode >= 300) {
      // Every non-2xx response from the write API has the shape { message }.
      String message;
      try {
        message = jsonDecode(response.body)['message'] as String;
      } catch (_) {
        message = response.reasonPhrase ?? '';
      }
      if (response.statusCode == 401 || response.statusCode == 403) {
        throw AuthenticationException('Write API rejected the token (${response.statusCode}): $message');
      }
      throw Exception('Write API request failed (${response.statusCode}): $message');
    }

    // One result per transaction sent. A result's status is 'success', 'retryable_error', 'fatal_error' or
    // 'not_attempted'. A fatal error has requires_client_handling and failed_operation.error_code.
    final results = jsonDecode(response.body)['results'];
    if (results is! List || results.length != 1) {
      throw Exception('Expected one result for one transaction');
    }
    return Map<String, dynamic>.from(results.first as Map);
  }

  /// Turns the retryMaxAttempts-th consecutive retryable error for the same transaction into a fatal error for
  /// onFatalTransaction(). The write API retries database errors that it doesn't recognize as permanent, so
  /// without a limit, one bad transaction could block the queue forever.
  Map<String, dynamic> _applyRetryLimit(CrudTransaction transaction, Map<String, dynamic> result) {
    if (result['status'] != 'retryable_error') {
      _retryKey = null;
      return result;
    }
    final key = transaction.transactionId ?? transaction.crud.first.clientId;
    _retryCount = key == _retryKey ? _retryCount + 1 : 1;
    _retryKey = key;
    if (_retryCount < retryMaxAttempts) {
      return result;
    }
    _retryCount = 0;
    return {
      'status': 'fatal_error',
      'message': result['message'],
      'requires_client_handling': true,
      'failed_operation': {'error_code': 'RETRY_LIMIT_EXCEEDED', 'message': result['message']},
    };
  }
}

enum FatalDecision { retain, complete }

/// Thrown when the write API rejects the token (401 or 403).
class AuthenticationException implements Exception {
  AuthenticationException(this.message);

  final String message;

  @override
  String toString() => message;
}
```
